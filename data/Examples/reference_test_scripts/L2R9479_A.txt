def input_verification(sbdv_target):
    wcr_ptc.verify_state({"OB_State": 'ACTIVE', "expected_result": True,
                          "req_verified": 'L2R9479'})
    # subdiv target generates both when subdiv target is present and for TBC137
    wcr_ptc.verify_target_details(
        {"TGT_TYPE": sbdv_target[0] + " SPEED RESTRICTION", "TGT_START_LOC_BLOCKID": sbdv_target[1],
         "TGT_END_LOC_BLOCKID": sbdv_target[2],
         "TGT_START_LOC_OFFSET": sbdv_target[3], "TGT_END_LOC_OFFSET": sbdv_target[4],
         "TGT_START_LOC_DISTID": 8880, "TGT_SPEED_LIMIT": sbdv_target[5],
         "TGT_SPEED_APPLICABILITY": 0, "TGT_RESTRICTED_SPEED": 1, "expected_result": True,
         "req_verified": 'L2R9479_track level target verification'})
    


def output_verification(img):
    wcr_display.compare_image({"image": img, "expected_result": "DISPLAYED",
                               "req_verified": "L2R9479_most restrictive target label verification"})
    if case == 9:
        wcr_debug_client.log_verification({'keyword': "UP|8880|1001|0|UP|8880|1015|2483|SUBDIV SPEED RESTRIC|16|0|1",
                                           'keyword_2': "UP|8880|1001|0|UP|8880|1015|2483|SUBDIV SPEED RESTRIC|16|0|1",
                                       'expected_result': 'True', 'req_verified':"L2R9479_A: Verify Identical "
                                            "duplicate targets are created"})


def send_1051_msg(case):
    wcr_office_01051.reset_msg()
    wcr_office_01051.set_instance(0)
    wcr_office_01051.set_reason('New Authority')
    wcr_office_01051.set_type('Track warrant')
    wcr_office_01051.set_num(100)
    wcr_office_01051.set_sub_id(8880)
    wcr_office_01051.segment_set_segment('Bidirectional')
    if case not in [2, 6, 10, 14]:  # increasing
        wcr_office_01051.segment_set_from_to({"start": 50000, "end": 650000, "sub_id": 8880, "track": 'Main'})
    else:  # decreasing
        wcr_office_01051.segment_set_from_to({"start": 650000, "end": 30000, "sub_id": 8880, "track": 'Main'})
    wcr_office_01051.send_msg()
