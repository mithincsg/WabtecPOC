# User Defined Functions
def send_1041_msg_ptc():
    wcr_office_01040.set_num(108)
    wcr_office_01040.send_msg()
    wcr_office_base.reset_msg(2042)
    wcr_office_01041.set_instance(1)
    wcr_office_01041.set_type('PTC Suspension')
    wcr_office_01041.set_displayable_num(108)
    wcr_office_01041.set_num(108)
    wcr_office_01041.set_sub_id(8214)
    # send_1041_msg(180000, 201000, 201900, 203000)
    wcr_office_01041.set_segment(
        {"start": 180000, "end": 201000, "sub_id": 8214, "track": 'Main1', "all_track_flag": False})
    wcr_office_01041.set_segment(
        {"start": 201900, "end": 203000, "sub_id": 8214, "track": 'Main1', "all_track_flag": False})
    wcr_office_01041.set_text_line('PTC Suspension')
    wcr_office_01041.send_msg()


def send_1041_msg(startmp=None, endmp=None, startmp1=None, endmp1=None, effective=None, effectivetime=None):
    wcr_office_01041.set_instance('0')
    wcr_office_01041.set_displayable_num(101)
    wcr_office_01041.set_num(101)
    wcr_office_01041.set_type('Work zone')
    wcr_office_01041.set_sub_id(8108)
    # wcr_office_01041.set_restriction( {"head_end":headEnd,"restricted":False,"speed":0,"type":'Freight' } )
    wcr_office_01041.set_segment({"end": endmp, "start": startmp, "sub_id": 8108, "track": 'Main2'})
    wcr_office_01041.set_segment({"end": endmp1, "start": startmp1, "sub_id": 8108, "track": 'Main2'})
    if effective is True:
        wcr_office_01041.set_effective_timestamp({"minute": effectivetime, "direction": '+'})
    wcr_office_01041.send_msg()


def set_ob_to_cutout():
    wcr_office_01010.set_bos({"bos": 'UP'})
    wcr_office_01010.reset_msg()
    wcr_office_01010.set_reason('Command locomotive system state to Non-controlling')
    wcr_office_base.send_msg(1010)
    wcr_office_01010.set_reason('Confirm receipt of 02010 message')
